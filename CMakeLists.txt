cmake_minimum_required(VERSION 3.24)
project(Cutlery VERSION 0.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Multimedia Svg Test)
qt_standard_project_setup(REQUIRES 6.8)
set(CUTLERY_ONNXRUNTIME_DIR "$ENV{CUTLERY_ONNXRUNTIME_DIR}" CACHE PATH "ONNX Runtime release folder")
# A whisper.cpp whisper-cli build, for testing automatic captions (packaged separately).
set(CUTLERY_WHISPER_CLI "$ENV{CUTLERY_WHISPER_CLI}" CACHE FILEPATH "whisper-cli executable for tests")
# The AI pack's models folder, for tests of the real face models (eye contact).
set(CUTLERY_TEST_MODELS "$ENV{CUTLERY_TEST_MODELS}" CACHE PATH "AI pack models folder for tests")
add_library(cutlery_engine STATIC src/ExportProfiles.cpp src/ExportProfiles.h src/Playback.cpp src/Playback.h src/Thumbnails.cpp src/Thumbnails.h src/AiJobs.cpp src/AiJobs.h src/Captions.cpp src/Captions.h src/KeyboardShortcuts.cpp src/KeyboardShortcuts.h src/MediaAnalysis.cpp src/MediaAnalysis.h src/Scopes.cpp src/Scopes.h src/SoundLibrary.cpp src/SoundLibrary.h src/Interchange.cpp src/Interchange.h src/Project.cpp src/RenderGraph.cpp src/Editor.cpp src/Project.h src/RenderGraph.h src/Editor.h src/RationalTime.h)
target_include_directories(cutlery_engine PUBLIC src)
target_link_libraries(cutlery_engine PUBLIC Qt6::Core Qt6::Gui Qt6::Quick Qt6::Multimedia Qt6::Svg)
qt_add_executable(Cutlery src/main.cpp)
qt_add_resources(Cutlery app_icon PREFIX "/" FILES assets/cutlery.png)
if(WIN32)
 target_sources(Cutlery PRIVATE assets/app.rc)
 target_include_directories(Cutlery PRIVATE assets)
endif()
target_link_libraries(Cutlery PRIVATE cutlery_engine Qt6::QuickControls2 Qt6::Multimedia)
qt_add_qml_module(Cutlery URI Cutlery VERSION 1.0 QML_FILES qml/Main.qml qml/Timeline.qml)
set_target_properties(Cutlery PROPERTIES WIN32_EXECUTABLE TRUE RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
enable_testing()
qt_add_executable(cutlery_tests tests/test_engine.cpp)
target_link_libraries(cutlery_tests PRIVATE cutlery_engine Qt6::Test)
set_target_properties(cutlery_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_test(NAME engine COMMAND cutlery_tests)
target_compile_definitions(cutlery_tests PRIVATE CUTLERY_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
if(CUTLERY_ONNXRUNTIME_DIR)
 add_dependencies(cutlery_tests cutlery-ai)
 target_compile_definitions(cutlery_tests PRIVATE CUTLERY_AI_WORKER="$<TARGET_FILE:cutlery-ai>")
 if(CUTLERY_WHISPER_CLI)
  file(TO_CMAKE_PATH "${CUTLERY_WHISPER_CLI}" WHISPER_CLI_PATH)
  target_compile_definitions(cutlery_tests PRIVATE CUTLERY_WHISPER_CLI="${WHISPER_CLI_PATH}")
 endif()
 if(CUTLERY_TEST_MODELS)
  file(TO_CMAKE_PATH "${CUTLERY_TEST_MODELS}" TEST_MODELS_PATH)
  target_compile_definitions(cutlery_tests PRIVATE CUTLERY_TEST_MODELS="${TEST_MODELS_PATH}")
 endif()
endif()
# The offscreen platform does not see system fonts on Windows; point it at them so text renders
# as in the application.
set(CUTLERY_TEST_FONTS "")
if(WIN32)
 file(TO_CMAKE_PATH "$ENV{WINDIR}/Fonts" WINDOWS_FONTS)
 set(CUTLERY_TEST_FONTS ";QT_QPA_FONTDIR=${WINDOWS_FONTS}")
endif()
set_tests_properties(engine PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen${CUTLERY_TEST_FONTS}")
install(TARGETS Cutlery RUNTIME DESTINATION .)
# Optional AI worker: built when an ONNX Runtime release folder (include/, lib/) is given. A
# DirectML build (include/dml_provider_factory.h, lib/DirectML.dll) adds GPU inference.
if(CUTLERY_ONNXRUNTIME_DIR)
 find_library(ONNXRUNTIME_LIBRARY onnxruntime PATHS "${CUTLERY_ONNXRUNTIME_DIR}/lib" NO_DEFAULT_PATH REQUIRED)
 qt_add_executable(cutlery-ai worker/main.cpp worker/matte.cpp worker/upscale.cpp worker/transcribe.cpp worker/eyecontact.cpp worker/faces.cpp worker/face.h worker/common.h)
 target_include_directories(cutlery-ai PRIVATE "${CUTLERY_ONNXRUNTIME_DIR}/include")
 target_link_libraries(cutlery-ai PRIVATE Qt6::Core Qt6::Gui "${ONNXRUNTIME_LIBRARY}")
 set_target_properties(cutlery-ai PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" WIN32_EXECUTABLE FALSE)
 if(EXISTS "${CUTLERY_ONNXRUNTIME_DIR}/include/dml_provider_factory.h")
  target_compile_definitions(cutlery-ai PRIVATE CUTLERY_DIRECTML)
 endif()
 if(WIN32)
  file(GLOB ONNXRUNTIME_DLLS "${CUTLERY_ONNXRUNTIME_DIR}/lib/*.dll")
  add_custom_command(TARGET cutlery-ai POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${ONNXRUNTIME_DLLS} "$<TARGET_FILE_DIR:cutlery-ai>")
  target_compile_options(cutlery-ai PRIVATE /utf-8)
 else()
  set_target_properties(cutlery-ai PROPERTIES BUILD_RPATH "${CUTLERY_ONNXRUNTIME_DIR}/lib")
 endif()
endif()
if(MSVC)
 target_compile_options(cutlery_engine PRIVATE /W4 /utf-8)
 target_compile_options(Cutlery PRIVATE /utf-8)
else()
 target_compile_options(cutlery_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
qt_add_executable(cutlery_ui_tests tests/test_ui.cpp)
target_link_libraries(cutlery_ui_tests PRIVATE cutlery_engine Qt6::Test Qt6::QuickControls2 Qt6::Multimedia)
target_compile_definitions(cutlery_ui_tests PRIVATE CUTLERY_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
if(CUTLERY_ONNXRUNTIME_DIR)
 add_dependencies(cutlery_ui_tests cutlery-ai)
 target_compile_definitions(cutlery_ui_tests PRIVATE CUTLERY_AI_WORKER="$<TARGET_FILE:cutlery-ai>")
endif()
set_target_properties(cutlery_ui_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_test(NAME interface COMMAND cutlery_ui_tests)
set_tests_properties(interface PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software${CUTLERY_TEST_FONTS}")
if(MSVC)
 target_compile_options(cutlery_tests PRIVATE /utf-8)
 target_compile_options(cutlery_ui_tests PRIVATE /utf-8)
endif()
