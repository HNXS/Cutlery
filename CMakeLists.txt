cmake_minimum_required(VERSION 3.24)
project(Cutlery VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Multimedia Test)
qt_standard_project_setup(REQUIRES 6.8)
add_library(cutlery_engine STATIC src/Project.cpp src/RenderGraph.cpp src/Editor.cpp src/Project.h src/RenderGraph.h src/Editor.h src/Time.h)
target_include_directories(cutlery_engine PUBLIC src)
target_link_libraries(cutlery_engine PUBLIC Qt6::Core Qt6::Gui Qt6::Quick)
qt_add_executable(Cutlery src/main.cpp)
target_link_libraries(Cutlery PRIVATE cutlery_engine Qt6::QuickControls2 Qt6::Multimedia)
qt_add_qml_module(Cutlery URI Cutlery VERSION 1.0 QML_FILES qml/Main.qml)
set_target_properties(Cutlery PROPERTIES WIN32_EXECUTABLE TRUE RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
enable_testing()
qt_add_executable(cutlery_tests tests/test_engine.cpp)
target_link_libraries(cutlery_tests PRIVATE cutlery_engine Qt6::Test)
set_target_properties(cutlery_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_test(NAME engine COMMAND cutlery_tests)
set_tests_properties(engine PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
install(TARGETS Cutlery RUNTIME DESTINATION .)
if(MSVC)
 target_compile_options(cutlery_engine PRIVATE /W4 /utf-8)
 target_compile_options(Cutlery PRIVATE /utf-8)
else()
 target_compile_options(cutlery_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
